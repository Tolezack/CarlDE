cmake_minimum_required(VERSION 3.16)
project(BrainrotCode_StreamDemo VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(brainrot_tests tests/test_stream.cpp)
target_include_directories(brainrot_tests PRIVATE include)
target_link_libraries(brainrot_tests PRIVATE Threads::Threads)
add_executable(brainrot_biomes_tests tests/test_biomes.cpp)
target_include_directories(brainrot_biomes_tests PRIVATE include)
target_link_libraries(brainrot_biomes_tests PRIVATE Threads::Threads)
enable_testing()
add_test(NAME phase02_biomes COMMAND brainrot_biomes_tests)
add_test(NAME stream_core COMMAND brainrot_tests)
option(BRAINROT_BUILD_GAME "Build GLFW 3.3 voxel demo" ON)
if(BRAINROT_BUILD_GAME)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GLFW3 REQUIRED IMPORTED_TARGET glfw3)
    add_executable(brainrot_voxel src/main.cpp)
    target_include_directories(brainrot_voxel PRIVATE include)
    target_link_libraries(brainrot_voxel PRIVATE Threads::Threads PkgConfig::GLFW3 ${CMAKE_DL_LIBS})
    target_compile_definitions(brainrot_voxel PRIVATE BR_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
endif()
